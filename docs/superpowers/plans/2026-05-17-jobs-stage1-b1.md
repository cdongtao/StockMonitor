# Jobs Stage 1 — B1 市场统计数据入库 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 搭建 jobs/ 项目框架，实现 B1（东方财富涨停市场统计）数据每日自动入库 PostgreSQL，支持 3 个月历史回填。

**Architecture:** 适配器分层——`data_sources/eastmoney.py` 封装 HTTP 调用与分页，`tasks/sync_market.py` 调适配器拿数据后 upsert 到 DB，`main.py` 注册 APScheduler 定时任务并在启动时触发回填。

**Tech Stack:** Python 3.11, FastAPI, SQLAlchemy 2.x (async), PostgreSQL 16, Alembic, httpx, APScheduler 3.x, pydantic-settings, pytest-asyncio

```python
import asyncio
from logging.config import fileConfig

from sqlalchemy.ext.asyncio import async_engine_from_config
from sqlalchemy import pool

from alembic import context

from core.config import settings
from core.database import Base
import models.market_daily_stats  # noqa: F401 — 注册模型到 metadata

config = context.config
config.set_main_option("sqlalchemy.url", settings.database_url)

if config.config_file_name is not None:
    fileConfig(config.config_file_name)

target_metadata = Base.metadata


def run_migrations_offline() -> None:
    url = config.get_main_option("sqlalchemy.url")
    context.configure(url=url, target_metadata=target_metadata, literal_binds=True)
    with context.begin_transaction():
        context.run_migrations()


def do_run_migrations(connection):
    context.configure(connection=connection, target_metadata=target_metadata)
    with context.begin_transaction():
        context.run_migrations()


async def run_async_migrations() -> None:
    connectable = async_engine_from_config(
        config.get_section(config.config_ini_section, {}),
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )
    async with connectable.connect() as connection:
        await connection.run_sync(do_run_migrations)
    await connectable.dispose()


def run_migrations_online() -> None:
    asyncio.run(run_async_migrations())


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()
```

```python
# jobs/data_sources/base.py
import asyncio
import logging
from typing import Any

import httpx

from core.config import settings

logger = logging.getLogger(__name__)

BASE_URL = "https://datacenter.eastmoney.com/securities/api/data/v1/get"
EM_TIMESTAMP = "1770038173236"


class EastMoneyBase:
    def __init__(self):
        self.client = httpx.AsyncClient(timeout=settings.em_timeout)

    async def _fetch_page(
        self,
        report_name: str,
        columns: str,
        filter_str: str,
        page_number: int = 1,
    ) -> dict:
        params = {
            "source": "SECURITIES",
            "client": "APP",
            "reportName": report_name,
            "columns": columns,
            "filter": filter_str,
            "pageNumber": str(page_number),
            "pageSize": str(settings.em_page_size),
            "em_timestamp": EM_TIMESTAMP,
        }
        for attempt in range(settings.em_max_retry + 1):
            try:
                resp = await self.client.get(BASE_URL, params=params)
                resp.raise_for_status()
                return resp.json()
            except Exception as e:
                if attempt == settings.em_max_retry:
                    raise
                logger.warning(f"Retry {attempt + 1} for {report_name} page {page_number}: {e}")
                await asyncio.sleep(1)

    async def fetch_all_pages(
        self,
        report_name: str,
        columns: str,
        filter_str: str,
        **kwargs: Any,
    ) -> list[dict]:
        try:
            first = await self._fetch_page(report_name, columns, filter_str, page_number=1)
        except Exception as e:
            logger.error(f"Failed to fetch first page of {report_name}: {e}")
            return []

        result = first.get("result") or {}
        data = result.get("data") or []
        total_pages = result.get("pages", 1)

        if total_pages <= 1:
            return data

        # 并发请求剩余页
        semaphore = asyncio.Semaphore(settings.em_concurrent)

        async def fetch_page_safe(page: int) -> list[dict]:
            async with semaphore:
                try:
                    resp = await self._fetch_page(report_name, columns, filter_str, page_number=page)
                    return (resp.get("result") or {}).get("data") or []
                except Exception as e:
                    logger.error(f"Failed page {page} of {report_name}: {e}")
                    return []

        tasks = [fetch_page_safe(p) for p in range(2, total_pages + 1)]
        pages_data = await asyncio.gather(*tasks)

        for page_data in pages_data:
            data.extend(page_data)

        return data

    async def close(self):
        await self.client.aclose()
```

```python
from datetime import date

from data_sources.base import EastMoneyBase


class EastMoneyClient(EastMoneyBase):
    async def fetch_market_stats(self, trade_date: date) -> list[dict]:
        """B1: 涨停市场每日统计 — RPT_INTSELECTION_LIMITHIS"""
        columns = (
            "TRADE_DATE,LIMIT_NUMBERS,NATURAL_LIMIT,DAILY_LIMIT,TOUCH_LIMIT,"
            "SEALING_RATE,NATURAL_LIMIT_YES,LIMIT_PER_YES,POSITION_SUGGESTION,"
            "MONEYMAKING_EFFECT,SEALING_RATE_YES,LIMIT_DOWN_NUM,CJDT_NUM,DT_FBL"
        )
        filter_str = f"(TRADE_DATE='{trade_date.strftime('%Y-%m-%d')}')"
        return await self.fetch_all_pages(
            report_name="RPT_INTSELECTION_LIMITHIS",
            columns=columns,
            filter_str=filter_str,
        )

    async def fetch_trade_dates(self, limit: int = 100) -> list[str]:
        """获取最近交易日列表，用于回填判断"""
        data = await self.fetch_all_pages(
            report_name="RPT_INTSELECTION_DATELIST",
            columns="TRADE_DATE,IS_NEW",
            filter_str="",
        )
        return [row["TRADE_DATE"] for row in data[:limit] if row.get("TRADE_DATE")]
```

```python
# jobs/tasks/sync_market.py
import logging
from datetime import date, datetime

from sqlalchemy import text
from sqlalchemy.ext.asyncio import AsyncSession

from core.database import AsyncSessionFactory
from data_sources.eastmoney import EastMoneyClient

logger = logging.getLogger(__name__)


def _parse_trade_date(raw: str) -> date:
    return datetime.strptime(raw[:10], "%Y-%m-%d").date()


async def sync_market_stats(trade_date: date, session: AsyncSession | None = None) -> int:
    """同步 B1 市场统计数据，返回写入条数"""
    client = EastMoneyClient()
    try:
        rows = await client.fetch_market_stats(trade_date)
    finally:
        await client.close()

    if not rows:
        logger.warning(f"[B1] No data for {trade_date}")
        return 0

    upsert_sql = text("""
        INSERT INTO market_daily_stats (
            trade_date, limit_numbers, natural_limit, daily_limit, touch_limit,
            sealing_rate, limit_down_num, moneymaking_effect, position_suggestion, raw_data
        ) VALUES (
            :trade_date, :limit_numbers, :natural_limit, :daily_limit, :touch_limit,
            :sealing_rate, :limit_down_num, :moneymaking_effect, :position_suggestion, :raw_data::jsonb
        )
        ON CONFLICT (trade_date) DO UPDATE SET
            limit_numbers = EXCLUDED.limit_numbers,
            natural_limit = EXCLUDED.natural_limit,
            daily_limit = EXCLUDED.daily_limit,
            touch_limit = EXCLUDED.touch_limit,
            sealing_rate = EXCLUDED.sealing_rate,
            limit_down_num = EXCLUDED.limit_down_num,
            moneymaking_effect = EXCLUDED.moneymaking_effect,
            position_suggestion = EXCLUDED.position_suggestion,
            raw_data = EXCLUDED.raw_data
    """)

    import json

    params = []
    for row in rows:
        try:
            params.append({
                "trade_date": _parse_trade_date(row["TRADE_DATE"]),
                "limit_numbers": row.get("LIMIT_NUMBERS"),
                "natural_limit": row.get("NATURAL_LIMIT"),
                "daily_limit": row.get("DAILY_LIMIT"),
                "touch_limit": row.get("TOUCH_LIMIT"),
                "sealing_rate": row.get("SEALING_RATE"),
                "limit_down_num": row.get("LIMIT_DOWN_NUM"),
                "moneymaking_effect": row.get("MONEYMAKING_EFFECT"),
                "position_suggestion": row.get("POSITION_SUGGESTION"),
                "raw_data": json.dumps(row, ensure_ascii=False),
            })
        except Exception as e:
            logger.error(f"[B1] Failed to parse row {row}: {e}")

    if not params:
        return 0

    own_session = session is None
    if own_session:
        session = AsyncSessionFactory()

    try:
        await session.execute(upsert_sql, params)
        await session.commit()
        logger.info(f"[B1] Upserted {len(params)} rows for {trade_date}")
        return len(params)
    except Exception as e:
        await session.rollback()
        logger.error(f"[B1] DB error for {trade_date}: {e}")
        raise
    finally:
        if own_session:
            await session.close()
```

```python
# jobs/main.py
import asyncio
import logging
from datetime import datetime, timedelta

from apscheduler.schedulers.asyncio import AsyncIOScheduler

from core.config import settings
from core.database import AsyncSessionFactory
from data_sources.eastmoney import EastMoneyClient
from tasks.sync_market import sync_market_stats

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(name)s: %(message)s",
)
logger = logging.getLogger(__name__)


async def get_trade_dates(days: int) -> list[str]:
    """获取最近 days 天内的交易日列表"""
    client = EastMoneyClient()
    try:
        dates = await client.fetch_trade_dates(limit=days + 20)
        return dates[:days]
    finally:
        await client.close()


async def check_and_backfill() -> None:
    """启动时检查是否需要回填历史数据"""
    from sqlalchemy import text

    async with AsyncSessionFactory() as session:
        result = await session.execute(text("SELECT COUNT(*) FROM market_daily_stats"))
        count = result.scalar()

    if count and count > 0:
        logger.info(f"DB already has {count} records, skipping backfill")
        return

    logger.info(f"DB empty, starting backfill for {settings.backfill_days} days...")
    trade_dates = await get_trade_dates(settings.backfill_days)

    for date_str in reversed(trade_dates):  # 从旧到新
        try:
            trade_date = datetime.strptime(date_str[:10], "%Y-%m-%d").date()
            count = await sync_market_stats(trade_date)
            logger.info(f"Backfill {trade_date}: {count} rows")
        except Exception as e:
            logger.error(f"Backfill failed for {date_str}: {e}")
        await asyncio.sleep(settings.sync_delay)

    logger.info("Backfill complete")


async def run_daily_sync() -> None:
    """每日定时任务：同步今日数据"""
    today = datetime.now().date()
    logger.info(f"Daily sync started for {today}")
    try:
        count = await sync_market_stats(today)
        logger.info(f"Daily sync done: {count} rows")
    except Exception as e:
        logger.error(f"Daily sync failed: {e}")


async def main() -> None:
    await check_and_backfill()

    scheduler = AsyncIOScheduler()
    scheduler.add_job(run_daily_sync, "cron", hour=15, minute=35)
    # 后续阶段逐步添加：
    # scheduler.add_job(sync_limit_stocks_task, "cron", hour=15, minute=40)  # B2
    # scheduler.add_job(sync_monitor_pool_task, "cron", hour=15, minute=45)  # B4

    scheduler.start()
    logger.info("Scheduler started. Waiting for jobs...")

    try:
        await asyncio.Event().wait()
    except (KeyboardInterrupt, SystemExit):
        scheduler.shutdown()
        logger.info("Scheduler stopped")


if __name__ == "__main__":
    asyncio.run(main())
```
